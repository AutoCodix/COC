cmake_minimum_required(VERSION 3.20)
project(GranuleEffects LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(GranuleEffects WIN32 src/main.cpp)
target_compile_definitions(GranuleEffects PRIVATE UNICODE _UNICODE)
target_link_libraries(GranuleEffects PRIVATE user32 gdi32 comctl32 dwmapi shell32)
if(MSVC)
  target_compile_options(GranuleEffects PRIVATE /W4 /EHsc /permissive-)
endif()
